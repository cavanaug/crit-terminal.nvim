# crit-nvim Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a LazyVim Neovim plugin that reviews Crit plan/markdown sessions natively (Snacks layout, inline annotations + comments pane), talking only to Crit’s local HTTP API, with a `crit-nvim` shell entrypoint.

**Architecture:** Thin Lua HTTP client over Crit’s daemon (`crit status --json` → port; `POST /api/file/comments?path=`; `POST /api/finish`; poll/SSE for sync). Snacks layout for plan mode first; code mode skeleton last. Crit remains source of truth — plugin never writes review JSON.

**Tech Stack:** Neovim Lua, Snacks.nvim layouts, Crit CLI ≥0.18, optional render-markdown.nvim, `curl` via `vim.system` for HTTP, headless `nvim` for unit tests.

**Spec:** `docs/superpowers/specs/2026-07-20-crit-nvim-design.md`

-- lua/crit/client.lua
local M = {}

---@class crit.Client
---@field base_url string
local Client = {}
Client.__index = Client

---@param base_url string
function M.new(base_url)
  return setmetatable({ base_url = base_url:gsub("/$", "") }, Client)
end

--- Overridable for tests.
function M._request(method, url, body)
  local args = { "curl", "-sS", "-w", "\n%{http_code}", "-X", method, url, "-H", "Accept: application/json" }
  if body then
    table.insert(args, "-H")
    table.insert(args, "Content-Type: application/json")
    table.insert(args, "-d")
    table.insert(args, body)
  end
  local result = vim.system(args, { text = true }):wait()
  if result.code ~= 0 then
    return 0, result.stderr or "curl failed"
  end
  local out = result.stdout or ""
  local code = tonumber(out:match("\n(%d%d%d)$")) or 0
  local payload = out:gsub("\n%d%d%d$", "")
  return code, payload
end

local function request(self, method, path, body_tbl)
  local url = self.base_url .. path
  local body = body_tbl and vim.json.encode(body_tbl) or nil
  local status, payload = M._request(method, url, body)
  if status < 200 or status >= 300 then
    error(string.format("crit %s %s -> %d: %s", method, path, status, payload))
  end
  if payload == "" then
    return {}
  end
  return vim.json.decode(payload)
end

function Client:health()
  local status = select(1, M._request("GET", self.base_url .. "/api/health"))
  return status == 200
end

--- Poll until session ready (Crit returns 503 until SetSession).
function Client:wait_ready(timeout_ms)
  timeout_ms = timeout_ms or 15000
  local deadline = vim.uv.now() + timeout_ms
  while vim.uv.now() < deadline do
    local status, payload = M._request("GET", self.base_url .. "/api/session")
    if status == 200 then
      return vim.json.decode(payload)
    end
    if status ~= 503 then
      error("wait_ready: " .. status .. " " .. tostring(payload))
    end
    vim.uv.sleep(100)
  end
  error("crit session not ready within timeout")
end

function Client:session()
  return request(self, "GET", "/api/session")
end

function Client:files()
  return request(self, "GET", "/api/files/list")
end

function Client:list_file_comments(path)
  return request(self, "GET", "/api/file/comments?path=" .. vim.uri_encode(path))
end

---@param path string
---@param start_line integer
---@param end_line integer
---@param body string
function Client:add_file_comment(path, start_line, end_line, body)
  return request(self, "POST", "/api/file/comments?path=" .. vim.uri_encode(path), {
    start_line = start_line,
    end_line = end_line,
    body = body,
  })
end

function Client:update_comment(path, id, body)
  return request(self, "PUT", "/api/comment/" .. id .. "?path=" .. vim.uri_encode(path), { body = body })
end

function Client:delete_comment(path, id)
  return request(self, "DELETE", "/api/comment/" .. id .. "?path=" .. vim.uri_encode(path))
end

function Client:finish()
  return request(self, "POST", "/api/finish")
end

-- expose for tests
M._Client = Client
return M
```

-- lua/crit/session.lua
local client_mod = require("crit.client")
local config = require("crit.config")
local state = require("crit.state")

local M = {
  ---@type crit.Client|nil
  client = nil,
  mode = nil, -- "plan" | "code"
}

function M.parse_status_json(s)
  return vim.json.decode(s)
end

function M.base_url_from_status(status)
  local port = status.daemon and status.daemon.port
  if not port then
    return nil
  end
  return string.format("http://127.0.0.1:%d", port)
end

function M.detect_mode(opts)
  local path = opts.file or opts.path
  if path and path:match("%.md$") then
    return "plan"
  end
  return "code"
end

function M.crit_on_path()
  return vim.fn.executable("crit") == 1
end

function M.status()
  local result = vim.system({ "crit", "status", "--json" }, { text = true }):wait()
  if result.code ~= 0 then
    error("crit status failed: " .. (result.stderr or result.stdout or ""))
  end
  return M.parse_status_json(result.stdout)
end

--- Ensure a Crit daemon is up. Launch with --no-open if needed.
---@param opts { file?: string, base_url?: string }
function M.ensure_daemon(opts)
  opts = opts or {}
  if opts.base_url or config.get().base_url then
    return opts.base_url or config.get().base_url
  end
  if not M.crit_on_path() then
    error("crit not found on PATH — install from https://crit.md")
  end
  local st = M.status()
  if st.daemon and st.daemon.running and st.daemon.port then
    return M.base_url_from_status(st)
  end
  local args = { "crit", "--no-open" }
  if opts.file then
    table.insert(args, opts.file)
  end
  local result = vim.system(args, { text = true, detach = false }):wait()
  -- crit may daemonize; re-query status
  local deadline = vim.uv.now() + 10000
  while vim.uv.now() < deadline do
    st = M.status()
    if st.daemon and st.daemon.running and st.daemon.port then
      return M.base_url_from_status(st)
    end
    vim.uv.sleep(100)
  end
  error("failed to start crit daemon: " .. (result.stderr or result.stdout or ""))
end

---@param opts? { file?: string, base_url?: string }
function M.review(opts)
  opts = opts or {}
  if not opts.file and vim.bo.filetype == "markdown" and vim.api.nvim_buf_get_name(0) ~= "" then
    opts.file = vim.api.nvim_buf_get_name(0)
  end
  local base = M.ensure_daemon(opts)
  local client = client_mod.new(base)
  client:wait_ready()
  M.client = client
  M.mode = M.detect_mode(opts)
  state.load_from_client(client, opts.file)
  if M.mode == "plan" then
    require("crit.ui.plan").open(opts.file)
  else
    require("crit.ui.code").open()
  end
end

function M.finish()
  if not M.client then
    error("no active Crit session")
  end
  local res = M.client:finish()
  require("crit.ui.status").set("finished")
  return res
end

return M
```

-- lua/crit/ui/plan.lua
local state = require("crit.state")
local annotations = require("crit.annotations")
local config = require("crit.config")

local M = {
  layout = nil,
  doc_buf = nil,
  comments_buf = nil,
  status_buf = nil,
}

local function ensure_snacks()
  local ok, snacks = pcall(require, "snacks")
  if not ok or not snacks.layout then
    error("snacks.nvim layout required for crit-nvim")
  end
  return snacks
end

local function render_comments(buf)
  local lines = { "Comments", "────────" }
  for _, c in ipairs(state.comments) do
    local loc = string.format("L%d", c.start_line or 0)
    if c.end_line and c.end_line ~= c.start_line then
      loc = string.format("L%d-%d", c.start_line, c.end_line)
    end
    table.insert(lines, string.format("%s — %s", loc, (c.body or ""):gsub("\n", " ")))
  end
  if #state.comments == 0 then
    table.insert(lines, "(none)")
  end
  vim.bo[buf].modifiable = true
  vim.api.nvim_buf_set_lines(buf, 0, -1, false, lines)
  vim.bo[buf].modifiable = false
end

local function open_document(path)
  vim.cmd.edit(vim.fn.fnameescape(path))
  local buf = vim.api.nvim_get_current_buf()
  if config.get().render_markdown and pcall(require, "render-markdown") then
    pcall(vim.cmd, "RenderMarkdown buffer")
  end
  annotations.apply(buf, state.comments)
  return buf
end

function M.refresh()
  if M.comments_buf and vim.api.nvim_buf_is_valid(M.comments_buf) then
    render_comments(M.comments_buf)
  end
  if M.doc_buf and vim.api.nvim_buf_is_valid(M.doc_buf) then
    annotations.apply(M.doc_buf, state.comments)
  end
  require("crit.ui.status").render(M.status_buf)
end

function M.open(file)
  local Snacks = ensure_snacks()
  file = file or state.file
  if not file then
    error("no plan file for Crit workspace")
  end

  M.doc_buf = open_document(file)
  M.comments_buf = vim.api.nvim_create_buf(false, true)
  M.status_buf = vim.api.nvim_create_buf(false, true)
  vim.bo[M.comments_buf].filetype = "crit-comments"
  vim.bo[M.status_buf].filetype = "crit-status"
  render_comments(M.comments_buf)
  require("crit.ui.status").render(M.status_buf)

  M.layout = Snacks.layout.new({
    show = true,
    wins = {
      document = Snacks.win.new({
        show = false,
        buf = M.doc_buf,
        wo = { number = true, signcolumn = "yes" },
      }),
      comments = Snacks.win.new({
        show = false,
        buf = M.comments_buf,
        wo = { number = false, cursorline = true },
      }),
      status = Snacks.win.new({
        show = false,
        buf = M.status_buf,
        wo = { number = false },
      }),
    },
    layout = {
      box = "vertical",
      backdrop = false,
      width = 0,
      height = 0,
      {
        box = "horizontal",
        { win = "document", width = 0.7 },
        { win = "comments", width = 0.3 },
      },
      { win = "status", height = 1 },
    },
  })

  require("crit.keymaps").bind_workspace()
  require("crit.sync").start()
end

return M
```

Adjust `Snacks.win.new` / layout keys to match the installed Snacks API if the above constructors differ — read `snacks/layout.lua` examples in the local LazyVim install and mirror a known-working layout from Snacks docs/picker.

-- lua/crit/annotations.lua
local M = {
  ns = vim.api.nvim_create_namespace("crit_comments"),
  sign_group = "crit_comments",
}

vim.fn.sign_define("CritComment", { text = "┃", texthl = "DiagnosticInfo" })

function M.virt_text_for(comment)
  local body = (comment.body or ""):gsub("\n.*", "")
  return "💬 " .. body
end

function M.clear(buf)
  vim.api.nvim_buf_clear_namespace(buf, M.ns, 0, -1)
  vim.fn.sign_unplace(M.sign_group, { buffer = buf })
end

---@param buf integer
---@param comments table[]
function M.apply(buf, comments)
  M.clear(buf)
  for _, c in ipairs(comments or {}) do
    local line = math.max(0, (c.start_line or 1) - 1)
    vim.fn.sign_place(0, M.sign_group, "CritComment", buf, { lnum = line + 1 })
    vim.api.nvim_buf_set_extmark(buf, M.ns, line, 0, {
      virt_text = { { M.virt_text_for(c), "Comment" } },
      virt_text_pos = "eol",
    })
  end
end

return M

local function notify_err(err)
  vim.notify(tostring(err), vim.log.levels.ERROR)
end

function M.add_comment_on_selection()
  if not M.client then
    return notify_err("no active Crit session")
  end
  local path = require("crit.state").file
  if not path then
    return notify_err("no active file")
  end
  local start_line = vim.fn.line("v")
  local end_line = vim.fn.line(".")
  if start_line > end_line then
    start_line, end_line = end_line, start_line
  end
  -- normal mode: current line
  if vim.fn.mode() == "n" then
    start_line = vim.fn.line(".")
    end_line = start_line
  end
  vim.ui.input({ prompt = "Review Comment: " }, function(body)
    if not body or body == "" then
      return
    end
    local ok, comment = pcall(function()
      return M.client:add_file_comment(path, start_line, end_line, body)
    end)
    if not ok then
      return notify_err(comment)
    end
    require("crit.state").load_from_client(M.client, path)
    require("crit.ui.plan").refresh()
  end)
end

function M.comment_at_cursor()
  local path = require("crit.state").file
  local lnum = vim.fn.line(".")
  for _, c in ipairs(require("crit.state").comments or {}) do
    local a, b = c.start_line or 0, c.end_line or c.start_line or 0
    if lnum >= a and lnum <= b then
      return path, c
    end
  end
end

function M.delete_comment_at_cursor()
  local path, c = M.comment_at_cursor()
  if not c then
    return notify_err("no Crit comment on this line")
  end
  local ok, err = pcall(function()
    M.client:delete_comment(path, c.id)
  end)
  if not ok then
    return notify_err(err)
  end
  require("crit.state").load_from_client(M.client, path)
  require("crit.ui.plan").refresh()
end

function M.edit_comment_at_cursor()
  local path, c = M.comment_at_cursor()
  if not c then
    return notify_err("no Crit comment on this line")
  end
  vim.ui.input({ prompt = "Edit Comment: ", default = c.body or "" }, function(body)

-- lua/crit/navigation.lua
local state = require("crit.state")
local M = {}

local function sorted()
  local t = vim.deepcopy(state.comments or {})
  table.sort(t, function(a, b)
    return (a.start_line or 0) < (b.start_line or 0)
  end)
  return t
end

function M.next_comment()
  local list = sorted()
  local lnum = vim.fn.line(".")
  for _, c in ipairs(list) do
    if (c.start_line or 0) > lnum then
      vim.api.nvim_win_set_cursor(0, { c.start_line, 0 })
      return
    end
  end
  if list[1] then
    vim.api.nvim_win_set_cursor(0, { list[1].start_line, 0 })
  end
end

function M.prev_comment()
  local list = sorted()
  local lnum = vim.fn.line(".")
  for i = #list, 1, -1 do
    local c = list[i]
    if (c.start_line or 0) < lnum then
      vim.api.nvim_win_set_cursor(0, { c.start_line, 0 })
      return
    end
  end
  if list[#list] then
    vim.api.nvim_win_set_cursor(0, { list[#list].start_line, 0 })
  end
end

return M

-- lua/crit/sync.lua
local config = require("crit.config")
local state = require("crit.state")
local M = { timer = nil }

function M.start()
  M.stop()
  local ms = config.get().poll_ms
  M.timer = vim.uv.new_timer()
  M.timer:start(ms, ms, vim.schedule_wrap(function()
    local session = require("crit.session")
    if not session.client or not state.file then
      return
    end
    local ok, raw = pcall(function()
      return session.client:list_file_comments(state.file)
    end)
    if not ok then
      require("crit.ui.status").set("disconnected")
      require("crit.ui.plan").refresh()
      return
    end
    require("crit.ui.status").set("connected")
    local comments = raw.comments or raw or {}
    -- cheap equality: count + ids
    local function sig(list)
      local t = {}
      for _, c in ipairs(list) do
        table.insert(t, tostring(c.id) .. ":" .. tostring(c.body))
      end
      return table.concat(t, "|")
    end
    if sig(comments) ~= sig(state.comments) then
      state.set_comments(comments)
      require("crit.ui.plan").refresh()
    end
  end))
end

function M.stop()
  if M.timer then
    M.timer:stop()
    M.timer:close()
    M.timer = nil
  end
end

return M
```

```bash
#!/usr/bin/env bash
# scripts/crit-nvim — start/find Crit, open Neovim review workspace
set -euo pipefail

FILE="${1:-}"
if ! command -v crit >/dev/null; then
  echo "crit not found on PATH" >&2
  exit 1
fi
if ! command -v nvim >/dev/null; then
  echo "nvim not found on PATH" >&2
  exit 1
fi

# Launch or reuse daemon without browser
if [[ -n "$FILE" ]]; then
  crit --no-open "$FILE" >/dev/null &
else
  # attach path: if no daemon, crit with no files may still start git review
  if ! crit status --json 2>/dev/null | grep -q '"running": true'; then
    crit --no-open >/dev/null &
  fi
fi

# Wait for port
for i in $(seq 1 50); do
  PORT=$(crit status --json 2>/dev/null | sed -n 's/.*"port": *\([0-9]*\).*/\1/p' | head -1)
  if [[ -n "${PORT:-}" ]]; then
    break
  fi
  sleep 0.1
done
if [[ -z "${PORT:-}" ]]; then
  echo "crit daemon did not become ready" >&2
  exit 1
fi

URL="http://127.0.0.1:${PORT}"
# Prefer absolute file path for Neovim
ABS="${FILE}"
if [[ -n "$FILE" && "$FILE" != /* ]]; then
  ABS="$(pwd)/$FILE"
fi

export CRIT_NVIM_BASE_URL="$URL"
exec nvim \
  -c "lua require('crit').setup({ base_url = vim.env.CRIT_NVIM_BASE_URL })" \
  -c "lua require('crit').review({ file = [[${ABS}]], base_url = vim.env.CRIT_NVIM_BASE_URL })"
```
